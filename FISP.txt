?[
    this is a demonstration
    of a simple closure-based counter
]

(import [io] std.io)
(set [counter]
    (fn [x] (last
        (set [x] 0)
        (fn [] (last
            =[temp] x
            =[x] (+ x 1)
            temp
        ))
    )))

(last
    (set [c] (counter))
    (io.println "{}" (c)) ?? prints 0
    (io.println "{}" (c)) ?? prints 1
    (io.println "{}" (c)) ?? prints 2
    (io.println "{}" (c)) ?? prints 3
    (io.println "{}" (c)) ?? prints 4
    0
)
